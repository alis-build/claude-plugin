	gcppropagator "github.com/GoogleCloudPlatform/opentelemetry-operations-go/propagator"

	semconv "go.opentelemetry.io/otel/semconv/v1.26.0"
	"go.opentelemetry.io/otel/trace"
)

// tracerName identifies the instrumentation scope of spans this neuron creates
// itself, as opposed to the ones otelgrpc creates for each incoming RPC.
const tracerName = "marvel.sm.helloworld.v1"

// Tracer returns the tracer for spans created by this neuron's own code, e.g.
// ctx, span := Tracer().Start(ctx, "lookupUser"); defer span.End()
func Tracer() trace.Tracer {
	return otel.Tracer(tracerName)
}

// initTracing configures the global tracer provider to export spans to Cloud
// Trace in the neuron's own project. The returned function flushes buffered
// spans and must be called before the process exits.
//
// Tracing is skipped when ALIS_OS_PROJECT is unset, which is the case for a
// local run outside the alis environment. The global no-op tracer stays in
// place there, so instrumented code keeps working without credentials.
func initTracing(ctx context.Context) (func(context.Context) error, error) {
	noop := func(context.Context) error { return nil }

	projectID := os.Getenv("ALIS_OS_PROJECT")
	if projectID == "" {
		alog.Debug(ctx, "ALIS_OS_PROJECT is not set, tracing is disabled")
		return noop, nil
	}

	exporter, err := texporter.New(texporter.WithProjectID(projectID))
	if err != nil {
		return noop, err
	}

	// Resource detection records which Cloud Run service, revision and region a
	// span came from. Off GCP the detector fails and returns a partial resource,
	// which is still worth keeping.
	res, err := resource.New(ctx,
		resource.WithDetectors(gcp.NewDetector()),
		resource.WithTelemetrySDK(),
		resource.WithAttributes(
			semconv.ServiceNameKey.String(serviceName()),
			semconv.ServiceVersionKey.String(os.Getenv("K_REVISION")),
		),
	)
	if err != nil {
		alog.Warnf(ctx, "partial otel resource detection: %v", err)
	}
	if res == nil {
		res = resource.Default()
	}

	tp := sdktrace.NewTracerProvider(
		sdktrace.WithBatcher(exporter),
		sdktrace.WithResource(res),
		// ParentBased keeps a trace whole: once an upstream caller decides to
		// sample, every downstream neuron in that trace samples too.
		sdktrace.WithSampler(sdktrace.ParentBased(sdktrace.TraceIDRatioBased(sampleRatio(ctx)))),
	)
	otel.SetTracerProvider(tp)

	// Accept both the W3C traceparent header used between neurons and the
	// X-Cloud-Trace-Context header Google's load balancers add on ingress.
	otel.SetTextMapPropagator(propagation.NewCompositeTextMapPropagator(
		gcppropagator.CloudTraceOneWayPropagator{},
		propagation.TraceContext{},
		propagation.Baggage{},
	))

	return tp.Shutdown, nil
}

// serviceName is the Cloud Run service name, which alis sets to the neuron name.
func serviceName() string {
	if name := os.Getenv("K_SERVICE"); name != "" {
		return name
	}
	return tracerName
}

// sampleRatio is the fraction of traces to record, from OTEL_TRACE_SAMPLE_RATIO.
// Local runs record everything; on Cloud Run the default matches cloudrun.tf.
func sampleRatio(ctx context.Context) float64 {
	raw := os.Getenv("OTEL_TRACE_SAMPLE_RATIO")
	if raw == "" {
		if os.Getenv("K_SERVICE") == "" {
			return 1.0
		}
		return 0.1
	}
	ratio, err := strconv.ParseFloat(raw, 64)
	if err != nil {
		alog.Warnf(ctx, "OTEL_TRACE_SAMPLE_RATIO %q is not a number, sampling all traces", raw)
		return 1.0
	}
	return ratio
}

	port := ":8080"
	listener, err := net.Listen("tcp", port)
	if err != nil {
		alog.Fatalf(ctx, "net.Listen: %v", err.Error())
	}

	// Cloud Run sends SIGTERM before tearing an instance down. Drain in-flight
	// RPCs so their spans finish, then flush what the batcher is holding.
	stop := make(chan os.Signal, 1)
	signal.Notify(stop, syscall.SIGTERM, syscall.SIGINT)
	go func() {
		<-stop
		alog.Notice(ctx, "shutting down")
		grpcServer.GracefulStop()
	}()

	alog.Notice(ctx, "starting server on "+port)
	if err = grpcServer.Serve(listener); err != nil {
		alog.Fatal(ctx, err.Error())
	}

	flushCtx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
	defer cancel()
	if err := shutdownTracing(flushCtx); err != nil {
		alog.Warnf(ctx, "flushing traces: %v", err)
	}
}

// Unary interceptor that logs failures and marks them on the request's span.
func unaryInterceptor(ctx context.Context, req interface{}, info *grpc.UnaryServerInfo, handler grpc.UnaryHandler) (interface{}, error) {

	// Call the handler
	h, err := handler(ctx, req)
	if err != nil {
		alog.Debugf(ctx, "%s", req)
		alog.Warn(ctx, err.Error())

		span := trace.SpanFromContext(ctx)
		span.RecordError(err)
		span.SetStatus(codes.Error, err.Error())
	}
	return h, err
}
