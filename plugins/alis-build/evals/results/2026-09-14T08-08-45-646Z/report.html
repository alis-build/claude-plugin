type InvoiceCreatedEvent struct {

// PublishInvoiceCreated publishes an invoice creation event and blocks until
// the server acknowledges it, so a failed publish surfaces to the caller.
func PublishInvoiceCreated(ctx context.Context, event *InvoiceCreatedEvent) error {
	pubsubOnce.Do(func() {
		pubsubClient, pubsubErr = pubsub.NewClient(ctx, os.Getenv("ALIS_OS_PROJECT"))
	})
	if pubsubErr != nil {
		return pubsubErr
	}

	if err != nil {
		return err

		Data:       data,
		Attributes: map[string]string{"event": "invoice-created"},
	})
	id, err := res.Get(ctx)
	if err != nil {
		return err
	}
	alog.Debugf(ctx, "published invoice-created event %s", id)
	return nil
}

import (
	"context"
	"os"
	"strconv"
	"time"

	texporter "github.com/GoogleCloudPlatform/opentelemetry-operations-go/exporter/trace"
	"go.alis.build/alog"

	"go.opentelemetry.io/otel/attribute"
	"go.opentelemetry.io/otel/propagation"
	"go.opentelemetry.io/otel/sdk/resource"
	sdktrace "go.opentelemetry.io/otel/sdk/trace"

// tracer is used for custom spans within this neuron. Spans for incoming grpc
// calls are created by the otelgrpc stats handler registered in server.go.
var tracer = otel.Tracer("marvel.sm.helloworld.v1")

// setupTracing exports spans to Google Cloud Trace and returns a shutdown func
// that flushes anything still buffered.
//
// If no project can be resolved (i.e. running locally) the global tracer is left
// as the default no-op so that the server still starts.
func setupTracing(ctx context.Context) func() {
	// Propagate trace context onto outgoing calls to other neurons.
	otel.SetTextMapPropagator(propagation.NewCompositeTextMapPropagator(
		propagation.TraceContext{},
		propagation.Baggage{},
	))

	projectID := os.Getenv("ALIS_OS_PROJECT")
	if projectID == "" {
		projectID = os.Getenv("GOOGLE_CLOUD_PROJECT")
	}
	if projectID == "" {
		alog.Warn(ctx, "ALIS_OS_PROJECT not set, tracing disabled")
		return func() {}
	}

	exporter, err := texporter.New(texporter.WithProjectID(projectID))
	if err != nil {
		// Tracing is not worth refusing to serve traffic over.
		alog.Errorf(ctx, "cloud trace exporter: %v", err)
		return func() {}
	}

	res, err := resource.Merge(resource.Default(), resource.NewWithAttributes(
		resource.Default().SchemaURL(),
		attribute.String("service.name", serviceName()),
		attribute.String("service.version", os.Getenv("K_REVISION")),
	))
	if err != nil {
		alog.Errorf(ctx, "build trace resource: %v", err)
		res = resource.Default()
	}

	provider := sdktrace.NewTracerProvider(
		sdktrace.WithResource(res),
		sdktrace.WithSampler(sdktrace.ParentBased(sdktrace.TraceIDRatioBased(samplingRatio(ctx)))),
		// Cloudrun only guarantees CPU during a request (cpu_idle = true), so keep
		// the batch window short and rely on the shutdown flush for the remainder.
		sdktrace.WithBatcher(exporter, sdktrace.WithBatchTimeout(5*time.Second)),
	)
	otel.SetTracerProvider(provider)

	return func() {
		shutdownCtx, cancel := context.WithTimeout(context.Background(), 10*time.Second)
		defer cancel()
		if err := provider.Shutdown(shutdownCtx); err != nil {
			alog.Errorf(shutdownCtx, "shutdown tracer provider: %v", err)
		}
	}
}

// serviceName returns the neuron name, falling back to the module name locally.
func serviceName() string {
	if name := os.Getenv("K_SERVICE"); name != "" {
		return name
	}
	return "marvel.sm.helloworld.v1"
}

// samplingRatio is the fraction of traces to record, set by OTEL_TRACE_SAMPLING_RATIO.
// Everything is sampled when not running on cloudrun.
func samplingRatio(ctx context.Context) float64 {
	if os.Getenv("K_SERVICE") == "" {
		return 1.0
	}
	ratio := 0.1
	if v := os.Getenv("OTEL_TRACE_SAMPLING_RATIO"); v != "" {
		parsed, err := strconv.ParseFloat(v, 64)
		if err != nil {
			alog.Warnf(ctx, "OTEL_TRACE_SAMPLING_RATIO must be a float, defaulting to %v", ratio)
		} else {
			ratio = parsed
		}
	}
	return ratio
}

### `server.go`

`main` gains the tracer provider and a SIGTERM handler — without the graceful stop, Cloud Run kills the instance and buffered spans are lost. Note `alog.Fatalf` on the `net.Listen` error still bypasses the deferred flush, which is fine since no spans exist yet; the `Serve` error becomes `alog.Error` so the flush runs.

import (
	"context"
	"net"
	"os"
	"os/signal"
	"strconv"
	"syscall"

	"go.alis.build/alog"
	"go.opentelemetry.io/contrib/instrumentation/google.golang.org/grpc/otelgrpc"
	"go.opentelemetry.io/otel/trace"
	"google.golang.org/grpc"
	pb "internal.sm.marvel.build/protobuf/marvel/sm/helloworld/v1"
)

func main() {
	ctx := context.Background()

	// Configure tracing before serving so startup is covered too.
	shutdownTracing := setupTracing(ctx)
	defer shutdownTracing()

	// Create grpc server and register implemented grpc services
	grpcServer := grpc.NewServer(
		grpc.StatsHandler(otelgrpc.NewServerHandler()),
		grpc.UnaryInterceptor(unaryInterceptor),
	)
	pb.RegisterHelloWorldServiceServer(grpcServer, HelloWorldServer)

	// Start the server
	port := ":8080"
	listener, err := net.Listen("tcp", port)
	if err != nil {
		alog.Fatalf(ctx, "net.Listen: %v", err.Error())
	}

	// Cloudrun sends SIGTERM before tearing an instance down: stop accepting new
	// requests so the deferred shutdown can flush buffered spans.
	stop := make(chan os.Signal, 1)
	signal.Notify(stop, syscall.SIGTERM, os.Interrupt)
	go func() {
		<-stop
		alog.Notice(ctx, "stopping server")
		grpcServer.GracefulStop()
	}()

	alog.Notice(ctx, "starting server on "+port)
	if err = grpcServer.Serve(listener); err != nil {
		alog.Error(ctx, err.Error())
	}
}
