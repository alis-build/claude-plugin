package dates

import (
	"errors"
	"strings"
	"testing"
	"time"
)

func TestParseDate(t *testing.T) {
	tests := []struct {
		name string
		in   string
		want time.Time
	}{
		{
			name: "ordinary date",
			in:   "2024-03-17",
			want: time.Date(2024, time.March, 17, 0, 0, 0, 0, time.UTC),
		},
		{
			name: "leap day",
			in:   "2024-02-29",
			want: time.Date(2024, time.February, 29, 0, 0, 0, 0, time.UTC),
		},
		{
			name: "first day of year",
			in:   "2021-01-01",
			want: time.Date(2021, time.January, 1, 0, 0, 0, 0, time.UTC),
		},
		{
			name: "last day of year",
			in:   "2021-12-31",
			want: time.Date(2021, time.December, 31, 0, 0, 0, 0, time.UTC),
		},
	}

	for _, tt := range tests {
		t.Run(tt.name, func(t *testing.T) {
			got, err := parseDate(tt.in)
			if err != nil {
				t.Fatalf("parseDate(%q) returned unexpected error: %v", tt.in, err)
			}
			if !got.Equal(tt.want) {
				t.Errorf("parseDate(%q) = %v, want %v", tt.in, got, tt.want)
			}
			// Equal only compares instants, so assert the zone and wall clock
			// that callers will actually format and store.
			if loc := got.Location(); loc != time.UTC {
				t.Errorf("parseDate(%q) location = %v, want UTC", tt.in, loc)
			}
			if h, m, s := got.Clock(); h != 0 || m != 0 || s != 0 || got.Nanosecond() != 0 {
				t.Errorf("parseDate(%q) time of day = %02d:%02d:%02d.%09d, want midnight",
					tt.in, h, m, s, got.Nanosecond())
			}
		})
	}
}

			}

func TestParseDateErrorWrapsParseError(t *testing.T) {
	const in = "not-a-date"

	_, err := parseDate(in)
	if err == nil {
		t.Fatalf("parseDate(%q) = nil error, want error", in)
	}

	// The message should name the function and echo the offending input so the
	// error is actionable without the caller re-reading the input.
	if msg, want := err.Error(), `parseDate "`+in+`"`; !strings.Contains(msg, want) {
		t.Errorf("parseDate(%q) error = %q, want it to contain %q", in, msg, want)
	}

	var parseErr *time.ParseError
	if !errors.As(err, &parseErr) {
		t.Errorf("parseDate(%q) error = %v, want it to wrap *time.ParseError", in, err)
	}
}

func TestParseDateRoundTrip(t *testing.T) {
	const in = "1999-12-31"

	got, err := parseDate(in)
	if err != nil {
		t.Fatalf("parseDate(%q) returned unexpected error: %v", in, err)
	}
	if formatted := got.Format("2006-01-02"); formatted != in {
		t.Errorf("parseDate(%q) formatted back to %q, want %q", in, formatted, in)
	}
}

What it covers:

- **Happy path** — value equality plus two things `Equal` alone won't catch: the returned `Location` is UTC, and the clock is exactly midnight.
- **Rejections** — wrong separators, wrong field order, unpadded components (`2006-01-02` uses fixed-width `01`/`02`, so `2024-3-7` is a parse failure), out-of-range month/day, non-leap Feb 29, surrounding whitespace, and an RFC 3339 string with a trailing time. Each also asserts the returned `time.Time` is zero rather than partially filled.
- **Error contract** — the `%q` input appears in the message and `%w` keeps `*time.ParseError` reachable through `errors.As`.

Two things worth flagging about the code under test: `t.UTC()` is a no-op here, since a layout with no zone already parses into UTC — harmless, but it may read as stronger normalization than it is. And the whitespace cases pass today purely because `time.Parse` is strict; if you'd rather accept trimmed input, that's a behavior decision to make before these tests pin it down.

What it pins down beyond the happy path:

- **Location, not just instant.** `time.Time.Equal` ignores location, so a `parseDate` that returned the same instant in `time.Local` would pass an `Equal` check alone. The explicit `Location() != time.UTC` assertion is what actually covers the doc comment's "midnight UTC". (As written, the `.UTC()` call is a no-op — `time.Parse` with a zone-less layout already yields UTC — but the assertion keeps that true if the layout ever gains a zone.)
- **The zero time on failure**, which callers may rely on.
- **Error contract**: the message includes the quoted input, and the underlying `*time.ParseError` survives the `%w` wrap, so callers can `errors.As` it.

Two cases worth confirming against your intent rather than just locking in: `"2024-3-7"` (unpadded) and `"2024-03-17T00:00:00Z"` both fail with the `2006-01-02` layout. If you'd rather accept either, that's a change to `parseDate`, and I'd move those rows to the success table.
